170struct CacheStats
171{
172 u64 reads = 0;
173 u64 writes = 0;
174 u64 hits = 0;
175 u64 misses = 0;
176 u64 evictions = 0;
177 u64 dirty_evictions = 0;
178 u64 total_latency = 0;
179
180 [[nodiscard]] double hit_rate() const noexcept
181 {
182 u64 total = hits + misses;
183 return total > 0
184 ? static_cast<double>(hits) / static_cast<double>(total)
185 : 0.0;
186 }
187
188 [[nodiscard]] double miss_rate() const noexcept
189 {
190 return 1.0 - hit_rate();
191 }
192
193 [[nodiscard]] double avg_latency() const noexcept
194 {
195 u64 accesses = reads + writes;
196 return accesses > 0
197 ? static_cast<double>(total_latency) / static_cast<double>(accesses)
198 : 0.0;
199 }
200
201 void reset() noexcept { *this = CacheStats{}; }
202};
