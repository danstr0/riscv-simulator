106 case CSRAddr::MSTATUS:
107 // Only MIE and MPIE are writable
108 mstatus_ = value & (MStatus::MIE | MStatus::MPIE);
109 break;
110 case CSRAddr::MIE:
111 mie_ = value & (MInterrupt::MSIE | MInterrupt::MTIE | MInterrupt::MEIE);
112 break;
113 case CSRAddr::MTVEC:
114 mtvec_ = value;
115 break;
116 case CSRAddr::MEPC:
117 mepc_ = value & ~1u; // Bit 0 always zero (IALIGN=32)
118 break;
119 case CSRAddr::MCAUSE:
120 mcause_ = value;
121 break;
122 case CSRAddr::MIP:
123 // Hardware sets MTIP/MEIP directly; all bits writable here for simplicity
124 mip_ = value & (MInterrupt::MSIE | MInterrupt::MTIE | MInterrupt::MEIE);
125 break;
126 default:
127 break;
