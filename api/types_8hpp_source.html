161template<unsigned B>
162[[nodiscard]] constexpr i32 sign_extend(u32 value) noexcept
163{
164 static_assert(B > 0 && B <= 32, "Bit width must be in [1, 32]");
165
166 if constexpr (B == 32)
167 return static_cast<i32>(value);
168 else
169 {
170 constexpr u32 sign_bit = 1u << (B - 1);
171 constexpr u32 mask = (1u << B) - 1;
172 value &= mask;
173 return static_cast<i32>((value ^ sign_bit) - sign_bit);
174 }
175}

178[[nodiscard]] constexpr u32 bits(u32 value, unsigned hi, unsigned lo) noexcept
179{
180 assert(hi >= lo && "bits(): hi must be >= lo");
181
182 unsigned width = hi - lo + 1;
183 u32 mask = (width >= 32) ? ~u32{0} : (1u << width) - 1;
184 return (value >> lo) & mask;
185}

188[[nodiscard]] constexpr u32 bit(u32 value, unsigned pos) noexcept
189{
190 return (value >> pos) & 1;
