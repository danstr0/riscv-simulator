47struct VType
48{
49 u32 sew = 32;
50 u32 lmul = 1;
51 bool vta = false;
52 bool vma = false;
53 bool vill = false;
54
59 [[nodiscard]] constexpr u32 encode() const noexcept
60 {
61 if (vill) return 1u << 31;
62
63 u32 vsew_field = 0;
64 switch(sew)
65 {
66 case 8: vsew_field = 0b000; break;
67 case 16: vsew_field = 0b001; break;
68 case 32: vsew_field = 0b010; break;
69 case 64: vsew_field = 0b011; break;
70 default: return 1u << 31; // invalid sew -> vill
71 }
72 return (static_cast<u32>(vma) << 7)
73 | (static_cast<u32>(vta) << 6)
74 | (vsew_field << 3);
75 }

81 static VType decode(u32 zimm)
82 {
83 VType vt;
84 u32 vsew = (zimm >> 3) & 0x7;
85 switch (vsew)
86 {
87 case 0b000: vt.sew = 8; break;
88 case 0b001: vt.sew = 16; break;
89 case 0b010: vt.sew = 32; break;
90 case 0b011: vt.sew = 64; break;
91 default: vt.vill = true; return vt;
92 }
93 if (vt.sew != 32) vt.vill = true;
94 vt.vta = (zimm >> 6) & 1;
95 vt.vma = (zimm >> 7) & 1;
96 vt.lmul = 1;
97 return vt;
98 }
