118struct DecodedInst
119{
120 Op op = Op::INVALID;
121 Format format = Format::R;
122
123 reg_idx_t rd = 0;

125 reg_idx_t rs2 = 0;
126 i32 imm = 0;

128 u32 raw = 0;

133 [[nodiscard]] constexpr bool writes_rd() const noexcept
134 {
135 switch (op)
136 {
137 case Op::SB: case Op::SH: case Op::SW:
138 case Op::BEQ: case Op::BNE: case Op::BLT:
139 case Op::BGE: case Op::BLTU: case Op::BGEU:
140 case Op::FENCE: case Op::ECALL: case Op::EBREAK:
141 case Op::INVALID:
142 return false;
143 default:
144 return rd != 0;
145 }
146 }
147
148 [[nodiscard]] constexpr bool reads_rs1() const noexcept
149 {
150 switch (op)
151 {
152 case Op::LUI: case Op::AUIPC: case Op::JAL:
153 case Op::FENCE: case Op::ECALL: case Op::EBREAK:
154 case Op::INVALID:
155 return false;
156 default:
157 return true;
158 }
159 }
160
161 [[nodiscard]] constexpr bool reads_rs2() const noexcept
162 {
163 return format == Format::R || format == Format::S || format == Format::B;
164 }
165
166 [[nodiscard]] constexpr bool is_load() const noexcept { return op >= Op::LB && op <= Op::LHU; }
167 [[nodiscard]] constexpr bool is_store() const noexcept { return op >= Op::SB && op <= Op::SW; }
168 [[nodiscard]] constexpr bool is_branch() const noexcept { return op >= Op::BEQ && op <= Op::BGEU; }
169 [[nodiscard]] constexpr bool is_jump() const noexcept { return op == Op::JAL || op == Op::JALR; }
170 [[nodiscard]] constexpr bool is_atomic() const noexcept { return op >= Op::LR_W && op <= Op::AMOMAXU_W; }
171 [[nodiscard]] constexpr bool is_vector() const noexcept { return op >= Op::VSETVLI && op <= Op::VMV_X_S; }
173
175 [[nodiscard]] std::string disassemble() const;
176};
